-- Users Table (Enhanced with soft delete and better indexing)
CREATE TABLE users (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    role ENUM('admin', 'freelancer', 'client') NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    phone VARCHAR(20),
    profile_picture VARCHAR(255),
    bio TEXT,
    country VARCHAR(50),
    city VARCHAR(50),
    address TEXT,
    postal_code VARCHAR(20),
    status ENUM('active', 'inactive', 'suspended', 'deleted') DEFAULT 'active',
    email_verified BOOLEAN DEFAULT FALSE,
    verification_token VARCHAR(255),
    verification_token_expires DATETIME,
    reset_token VARCHAR(255),
    reset_token_expires DATETIME,
    last_login DATETIME,
    last_activity DATETIME,
    failed_login_attempts TINYINT UNSIGNED DEFAULT 0,
    account_locked_until DATETIME,
    deleted_at DATETIME DEFAULT NULL,
    deleted_by INT UNSIGNED,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_user_role_status (role, status),
    INDEX idx_user_email_verified (email, email_verified),
    INDEX idx_user_created (created_at DESC),
    INDEX idx_user_last_activity (last_activity DESC),
    INDEX idx_user_deleted (deleted_at),
    INDEX idx_username_status (username, status),
    INDEX idx_verification_token (verification_token),
    INDEX idx_reset_token (reset_token),
    
    CONSTRAINT chk_email_format CHECK (email REGEXP '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$')
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci ROW_FORMAT=DYNAMIC;

-- Freelancer Profile (Enhanced with performance metrics)
CREATE TABLE freelancer_profiles (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id INT UNSIGNED NOT NULL UNIQUE,
    title VARCHAR(100),
    tagline VARCHAR(200),
    hourly_rate DECIMAL(10,2),
    availability_status ENUM('available', 'busy', 'unavailable') DEFAULT 'available',
    experience_level ENUM('beginner', 'intermediate', 'expert') DEFAULT 'beginner',
    total_earnings DECIMAL(15,2) DEFAULT 0.00,
    completed_jobs INT UNSIGNED DEFAULT 0,
    cancelled_jobs INT UNSIGNED DEFAULT 0,
    success_rate DECIMAL(5,2) DEFAULT 0.00,
    response_time INT UNSIGNED COMMENT 'Average response time in minutes',
    response_rate DECIMAL(5,2) DEFAULT 0.00,
    on_time_delivery_rate DECIMAL(5,2) DEFAULT 0.00,
    languages JSON COMMENT 'Array of {language, proficiency}',
    portfolio_url VARCHAR(255),
    video_intro_url VARCHAR(255),
    timezone VARCHAR(50),
    working_hours JSON COMMENT 'Working schedule',
    badges JSON COMMENT 'Achievement badges',
    profile_views INT UNSIGNED DEFAULT 0,
    profile_completeness TINYINT UNSIGNED DEFAULT 0,
    featured BOOLEAN DEFAULT FALSE,
    featured_until DATETIME,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_availability_rate (availability_status, hourly_rate),

    INDEX idx_profile_completeness (profile_completeness DESC),

-- Orders Table (Enhanced with better tracking)
CREATE TABLE orders (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    delivery_time INT UNSIGNED NOT NULL,
    extra_services JSON COMMENT 'Additional services purchased',
    requirements JSON COMMENT 'Client answers to requirements',
    special_instructions TEXT,
    priority ENUM('normal', 'high') DEFAULT 'normal',
    auto_complete_at DATETIME COMMENT 'Auto-complete if no action',
    started_at DATETIME,
    delivered_at DATETIME,
    completed_at DATETIME,
    cancelled_at DATETIME,
    cancellation_reason TEXT,
    cancelled_by INT UNSIGNED,
    dispute_reason TEXT,
    dispute_opened_at DATETIME,
    late_delivery BOOLEAN DEFAULT FALSE,
    revision_count INT UNSIGNED DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (client_id) REFERENCES users(id) ON DELETE RESTRICT,
    FOREIGN KEY (freelancer_id) REFERENCES users(id) ON DELETE RESTRICT,
    FOREIGN KEY (gig_id) REFERENCES gigs(id) ON DELETE RESTRICT,
    FOREIGN KEY (cancelled_by) REFERENCES users(id) ON DELETE SET NULL,
    
    INDEX idx_order_status_created (status, created_at DESC),
    INDEX idx_order_client_status (client_id, status),
    INDEX idx_order_freelancer_status (freelancer_id, status),
    INDEX idx_order_gig (gig_id),

-- Transactions Table (Enhanced with better tracking)
CREATE TABLE transactions (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    transaction_id VARCHAR(100) NOT NULL UNIQUE,
    order_id INT UNSIGNED,
    user_id INT UNSIGNED NOT NULL,
    transaction_type ENUM('payment', 'refund', 'withdrawal', 'deposit', 'service_fee', 'bonus', 'penalty') NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    currency VARCHAR(10) DEFAULT 'USD',
    status ENUM('pending', 'processing', 'completed', 'failed', 'cancelled', 'refunded') DEFAULT 'pending',
    payment_method ENUM('credit_card', 'paypal', 'stripe', 'bank_transfer', 'wallet', 'other') NOT NULL,
    payment_gateway VARCHAR(50),
    gateway_transaction_id VARCHAR(255),
    payment_gateway_response TEXT,
    description TEXT,
    metadata JSON,
    ip_address VARCHAR(45),
    processed_at DATETIME,
    failed_reason TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE SET NULL,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT,
    
    INDEX idx_transaction_user_type (user_id, transaction_type, status),
    INDEX idx_transaction_order (order_id),
    INDEX idx_transaction_status_created (status, created_at DESC),
    INDEX idx_transaction_gateway (payment_gateway, gateway_transaction_id),
    INDEX idx_created_at (created_at),  -- FIXED: Removed DATE() function
    
    CONSTRAINT chk_transaction_amount CHECK (amount != 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- User Wallets (Enhanced with locking mechanism)
CREATE TABLE wallets (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id INT UNSIGNED NOT NULL UNIQUE,
    balance DECIMAL(15,2) DEFAULT 0.00,
    pending_clearance DECIMAL(15,2) DEFAULT 0.00,
    reserved_amount DECIMAL(15,2) DEFAULT 0.00,
    total_earned DECIMAL(15,2) DEFAULT 0.00,
    total_withdrawn DECIMAL(15,2) DEFAULT 0.00,
    total_spent DECIMAL(15,2) DEFAULT 0.00,
    currency VARCHAR(10) DEFAULT 'USD',
    minimum_withdrawal DECIMAL(10,2) DEFAULT 50.00,
    last_transaction_at DATETIME,
    version INT UNSIGNED DEFAULT 0 COMMENT 'For optimistic locking',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT,
    INDEX idx_balance (balance DESC),
    INDEX idx_last_transaction (last_transaction_at DESC),
    
    CONSTRAINT chk_balance CHECK (balance >= 0),
    CONSTRAINT chk_pending CHECK (pending_clearance >= 0),
    CONSTRAINT chk_reserved CHECK (reserved_amount >= 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Wallet Transactions (Detailed audit trail)
CREATE TABLE wallet_transactions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    wallet_id INT UNSIGNED NOT NULL,
    transaction_id INT UNSIGNED,
    type ENUM('credit', 'debit', 'hold', 'release') NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    balance_before DECIMAL(15,2) NOT NULL,
    balance_after DECIMAL(15,2) NOT NULL,
    description VARCHAR(255),
    reference_type VARCHAR(50),
    reference_id INT UNSIGNED,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (wallet_id) REFERENCES wallets(id) ON DELETE CASCADE,
    FOREIGN KEY (transaction_id) REFERENCES transactions(id) ON DELETE SET NULL,
    INDEX idx_wallet_created (wallet_id, created_at DESC),
    INDEX idx_reference (reference_type, reference_id),
    
    CONSTRAINT chk_amount_positive CHECK (amount > 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Withdrawal Requests (Enhanced)
CREATE TABLE withdrawal_requests (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id INT UNSIGNED NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    withdrawal_method ENUM('bank_transfer', 'paypal', 'stripe', 'wise', 'other') NOT NULL,
    account_details TEXT NOT NULL COMMENT 'Encrypted account details',
    status ENUM('pending', 'processing', 'completed', 'rejected', 'cancelled') DEFAULT 'pending',
    admin_notes TEXT,
    rejection_reason TEXT,
    transaction_id INT UNSIGNED,
    processed_by INT UNSIGNED,
    processed_at DATETIME,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE RESTRICT,
    FOREIGN KEY (processed_by) REFERENCES users(id) ON DELETE SET NULL,
    FOREIGN KEY (transaction_id) REFERENCES transactions(id) ON DELETE SET NULL,
    
    INDEX idx_withdrawal_user_status (user_id, status),

DELIMITER //

-- Update gig statistics
CREATE PROCEDURE sp_update_gig_stats(IN p_gig_id INT)
BEGIN
    UPDATE gigs g
    SET 
        avg_rating = COALESCE((
            SELECT AVG(rating) 
            FROM reviews 
            WHERE gig_id = p_gig_id AND status = 'active'
        ), 0),
        reviews_count = COALESCE((
            SELECT COUNT(*) 
            FROM reviews 
            WHERE gig_id = p_gig_id AND status = 'active'
        ), 0)
    WHERE g.id = p_gig_id;
END//

-- Update freelancer statistics
CREATE PROCEDURE sp_update_freelancer_stats(IN p_freelancer_id INT)
BEGIN
    DECLARE v_completed_jobs INT;
    DECLARE v_total_orders INT;
    DECLARE v_total_earnings DECIMAL(15,2);
    DECLARE v_cancelled_jobs INT;
    DECLARE v_success_rate DECIMAL(5,2);
    
    SELECT 
        COUNT(CASE WHEN status = 'completed' THEN 1 END),
        COUNT(*),
        COALESCE(SUM(CASE WHEN status = 'completed' THEN amount END), 0),
        COUNT(CASE WHEN status = 'cancelled' THEN 1 END)
    INTO 
        v_completed_jobs, v_total_orders, v_total_earnings, v_cancelled_jobs
    FROM orders 
    WHERE freelancer_id = p_freelancer_id AND status IN ('completed', 'cancelled');
    
    IF v_total_orders > 0 THEN
        SET v_success_rate = (v_completed_jobs * 100.0 / v_total_orders);
    ELSE
        SET v_success_rate = 0;
    END IF;
    
    UPDATE freelancer_profiles
    SET 
        completed_jobs = v_completed_jobs,
        cancelled_jobs = v_cancelled_jobs,
        total_earnings = v_total_earnings,
        success_rate = v_success_rate
    WHERE user_id = p_freelancer_id;
END//

-- Update wallet balance (with locking)
CREATE PROCEDURE sp_update_wallet_balance(
    IN p_user_id INT,
    IN p_amount DECIMAL(10,2),
    IN p_type ENUM('credit', 'debit'),
    IN p_description VARCHAR(255),
    IN p_transaction_id INT
)
BEGIN
    DECLARE v_wallet_id INT;
    DECLARE v_balance_before DECIMAL(15,2);
    DECLARE v_balance_after DECIMAL(15,2);
    DECLARE v_version INT;
    
    START TRANSACTION;
    
    -- Lock the wallet row
    SELECT id, balance, version
    INTO v_wallet_id, v_balance_before, v_version
    FROM wallets
    WHERE user_id = p_user_id
    FOR UPDATE;
    
    -- Calculate new balance
    IF p_type = 'credit' THEN
        SET v_balance_after = v_balance_before + p_amount;
    ELSE
        SET v_balance_after = v_balance_before - p_amount;
    END IF;
    
    -- Ensure balance doesn't go negative
    IF v_balance_after < 0 THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Insufficient wallet balance';
    END IF;
    
    -- Update wallet
    UPDATE wallets
    SET balance = v_balance_after,
        version = version + 1,
        last_transaction_at = NOW()
    WHERE id = v_wallet_id AND version = v_version;
    
    IF ROW_COUNT() = 0 THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Wallet was modified by another transaction';
    END IF;
    
    -- Insert audit record
    INSERT INTO wallet_transactions (
        wallet_id, transaction_id, type, amount, 
        balance_before, balance_after, description
    ) VALUES (
        v_wallet_id, p_transaction_id, p_type, p_amount,
        v_balance_before, v_balance_after, p_description
    );
    
    COMMIT;
END//

-- Auto-complete orders after delivery period
CREATE EVENT IF NOT EXISTS evt_auto_complete_orders
ON SCHEDULE EVERY 1 HOUR
DO
BEGIN
    UPDATE orders
    SET status = 'completed',
        completed_at = NOW()
    WHERE status = 'delivered'
        AND auto_complete_at IS NOT NULL
        AND auto_complete_at <= NOW();
END//

-- Clean expired cache entries
CREATE EVENT IF NOT EXISTS evt_clean_cache
ON SCHEDULE EVERY 1 HOUR
DO
BEGIN
    CALL sp_clean_expired_cache();
END//

-- Update featured status
CREATE EVENT IF NOT EXISTS evt_update_featured_status
ON SCHEDULE EVERY 1 HOUR
DO
BEGIN
    -- Expire featured gigs
    UPDATE gigs
    SET featured = FALSE
    WHERE featured = TRUE
        AND featured_until IS NOT NULL
        AND featured_until <= NOW();
    
    -- Expire featured freelancers
    UPDATE freelancer_profiles
    SET featured = FALSE
    WHERE featured = TRUE
        AND featured_until IS NOT NULL
        AND featured_until <= NOW();
END//

-- Mark late deliveries
CREATE EVENT IF NOT EXISTS evt_mark_late_deliveries
ON SCHEDULE EVERY 1 HOUR
DO
BEGIN
    UPDATE orders
    SET late_delivery = TRUE,
        status = 'late'
    WHERE status = 'in_progress'
        AND started_at IS NOT NULL
        AND DATE_ADD(started_at, INTERVAL delivery_time DAY) < NOW();
END//

-- Update promo code status
CREATE EVENT IF NOT EXISTS evt_update_promo_status
ON SCHEDULE EVERY 1 DAY
DO
BEGIN
    UPDATE promo_codes
    SET status = 'expired'
    WHERE status = 'active'
        AND valid_until < NOW();
END//

-- Generate daily analytics
CREATE EVENT IF NOT EXISTS evt_generate_daily_analytics
ON SCHEDULE EVERY 1 DAY
STARTS CURRENT_DATE + INTERVAL 1 DAY
DO
BEGIN
    DECLARE v_date DATE DEFAULT CURDATE() - INTERVAL 1 DAY;
    
    -- Total orders by status
    INSERT INTO analytics_summary (date, metric_type, metric_value, metadata)
    SELECT 
        v_date,
        CONCAT('orders_', status),
        COUNT(*),
        JSON_OBJECT('status', status)
    FROM orders
    WHERE DATE(created_at) = v_date
    GROUP BY status
    ON DUPLICATE KEY UPDATE metric_value = VALUES(metric_value);
    
    -- Total revenue
    INSERT INTO analytics_summary (date, metric_type, metric_value)
    SELECT 
        v_date,
        'total_revenue',
        COALESCE(SUM(total_amount), 0)
    FROM orders
    WHERE DATE(created_at) = v_date AND status = 'completed'
    ON DUPLICATE KEY UPDATE metric_value = VALUES(metric_value);
    
    -- Active users
    INSERT INTO analytics_summary (date, metric_type, metric_value)
    SELECT 
        v_date,
        'active_users',
        COUNT(DISTINCT user_id)
    FROM activity_log
    WHERE DATE(created_at) = v_date
    ON DUPLICATE KEY UPDATE metric_value = VALUES(metric_value);
END//

-- Archive old data
CREATE EVENT IF NOT EXISTS evt_archive_old_data
ON SCHEDULE EVERY 1 WEEK
DO
BEGIN
    CALL sp_archive_old_logs(365);
END//
