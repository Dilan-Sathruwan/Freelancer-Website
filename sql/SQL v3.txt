-- ============================================================
-- FREELANCEHUB MARKETPLACE DATABASE - CLEAN DATA TABLES SCHEMA
-- Version: 3.0 (Optimized, Enhanced & Fixed)
-- ============================================================

SET FOREIGN_KEY_CHECKS = 0;

DROP DATABASE IF EXISTS freelancerwebsite;
CREATE DATABASE freelancerwebsite CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE freelancerwebsite;

-- ============================================================
-- 1. USERS & PROFILES
-- ============================================================

-- Users Table (Enhanced with soft delete, security tokens and performance indexes)
CREATE TABLE users (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    role ENUM('admin', 'freelancer', 'client') NOT NULL DEFAULT 'client',
    first_name VARCHAR(50) NOT NULL DEFAULT '',
    last_name VARCHAR(50) NOT NULL DEFAULT '',
    email VARCHAR(100) NOT NULL UNIQUE,
    phone VARCHAR(20) DEFAULT NULL,
    profile_picture VARCHAR(255) DEFAULT 'default.png',
    bio TEXT DEFAULT NULL,
    country VARCHAR(50) DEFAULT NULL,
    city VARCHAR(50) DEFAULT NULL,
    address TEXT DEFAULT NULL,
    postal_code VARCHAR(20) DEFAULT NULL,
    status ENUM('active', 'inactive', 'suspended', 'deleted') DEFAULT 'active',
    email_verified BOOLEAN DEFAULT FALSE,
    verification_token VARCHAR(255) DEFAULT NULL,
    verification_token_expires DATETIME DEFAULT NULL,
    reset_token VARCHAR(255) DEFAULT NULL,
    reset_token_expires DATETIME DEFAULT NULL,
    last_login DATETIME DEFAULT NULL,
    last_activity DATETIME DEFAULT NULL,
    failed_login_attempts TINYINT UNSIGNED DEFAULT 0,
    account_locked_until DATETIME DEFAULT NULL,
    deleted_at DATETIME DEFAULT NULL,
    deleted_by INT UNSIGNED DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_user_role_status (role, status),
    INDEX idx_user_email_verified (email, email_verified),
    INDEX idx_user_created (created_at DESC),
    INDEX idx_user_last_activity (last_activity DESC),
    INDEX idx_user_deleted (deleted_at),
    INDEX idx_username_status (username, status),
    INDEX idx_verification_token (verification_token),
    INDEX idx_reset_token (reset_token)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Freelancer Profiles Table
CREATE TABLE freelancer_profiles (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id INT UNSIGNED NOT NULL UNIQUE,
    title VARCHAR(100) DEFAULT NULL,
    tagline VARCHAR(200) DEFAULT NULL,
    hourly_rate DECIMAL(10,2) DEFAULT 0.00,
    availability_status ENUM('available', 'busy', 'unavailable') DEFAULT 'available',
    experience_level ENUM('beginner', 'intermediate', 'expert') DEFAULT 'beginner',
    total_earnings DECIMAL(15,2) DEFAULT 0.00,
    completed_jobs INT UNSIGNED DEFAULT 0,
    cancelled_jobs INT UNSIGNED DEFAULT 0,
    success_rate DECIMAL(5,2) DEFAULT 0.00,
    response_time INT UNSIGNED DEFAULT 60 COMMENT 'Average response time in minutes',
    response_rate DECIMAL(5,2) DEFAULT 100.00,
    on_time_delivery_rate DECIMAL(5,2) DEFAULT 100.00,
    languages JSON DEFAULT NULL COMMENT 'Array of {language, proficiency}',
    portfolio_url VARCHAR(255) DEFAULT NULL,
    video_intro_url VARCHAR(255) DEFAULT NULL,
    timezone VARCHAR(50) DEFAULT 'UTC',
    working_hours JSON DEFAULT NULL COMMENT 'Working schedule',
    badges JSON DEFAULT NULL COMMENT 'Achievement badges',
    profile_views INT UNSIGNED DEFAULT 0,
    profile_completeness TINYINT UNSIGNED DEFAULT 0,
    featured BOOLEAN DEFAULT FALSE,
    featured_until DATETIME DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_availability_rate (availability_status, hourly_rate),
    INDEX idx_success_rate (success_rate DESC),
    INDEX idx_response_time (response_time),
    INDEX idx_featured (featured, featured_until),
    INDEX idx_profile_completeness (profile_completeness DESC)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Client Profiles Table
CREATE TABLE client_profiles (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id INT UNSIGNED NOT NULL UNIQUE,
    company_name VARCHAR(100) DEFAULT NULL,
    company_website VARCHAR(255) DEFAULT NULL,
    company_size ENUM('1-10', '11-50', '51-200', '201-500', '500+') DEFAULT '1-10',
    industry VARCHAR(100) DEFAULT NULL,
    total_spent DECIMAL(15,2) DEFAULT 0.00,
    jobs_posted INT UNSIGNED DEFAULT 0,
    jobs_completed INT UNSIGNED DEFAULT 0,
    avg_budget DECIMAL(10,2) DEFAULT 0.00,
    payment_verified BOOLEAN DEFAULT FALSE,
    preferred_payment_method VARCHAR(50) DEFAULT NULL,
    verification_documents JSON DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_payment_verified (payment_verified),
    INDEX idx_total_spent (total_spent DESC)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Freelancers Table (Compatibility table for legacy queries)
CREATE TABLE freelancers (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id INT UNSIGNED DEFAULT NULL,
    username VARCHAR(50) NOT NULL,
    profile_picture VARCHAR(255) DEFAULT 'default.png',
    skills VARCHAR(255) DEFAULT NULL,
    hourly_rate DECIMAL(10,2) DEFAULT 0.00,
    location VARCHAR(100) DEFAULT NULL,
    rating DECIMAL(3,2) DEFAULT 5.00,
    bio TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_freelancers_user (user_id),
    INDEX idx_freelancers_rating (rating DESC)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================================
-- 2. CATEGORIES & SKILLS
-- ============================================================

-- Categories Table (With hierarchical support)
CREATE TABLE categories (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    slug VARCHAR(100) NOT NULL UNIQUE,
    description TEXT DEFAULT NULL,
    parent_id INT UNSIGNED DEFAULT NULL,
    level TINYINT UNSIGNED DEFAULT 0,
    path VARCHAR(255) DEFAULT NULL,
    icon VARCHAR(255) DEFAULT NULL,
    banner_image VARCHAR(255) DEFAULT NULL,
    display_order INT UNSIGNED DEFAULT 0,
    status ENUM('active', 'inactive') DEFAULT 'active',
    gigs_count INT UNSIGNED DEFAULT 0,
    meta_title VARCHAR(150) DEFAULT NULL,
    meta_description VARCHAR(255) DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (parent_id) REFERENCES categories(id) ON DELETE SET NULL,
    INDEX idx_category_slug_status (slug, status),
    INDEX idx_parent_level (parent_id, level),
    INDEX idx_status_order (status, display_order),
    INDEX idx_path (path)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Skills Table
CREATE TABLE skills (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE,
    slug VARCHAR(50) NOT NULL UNIQUE,
    category_id INT UNSIGNED DEFAULT NULL,
    description TEXT DEFAULT NULL,
    usage_count INT UNSIGNED DEFAULT 0,
    status ENUM('active', 'inactive') DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE SET NULL,
    INDEX idx_skill_slug_status (slug, status),
    INDEX idx_skill_category (category_id),
    INDEX idx_usage_count (usage_count DESC),
    FULLTEXT idx_skill_name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Freelancer Skills Relation Table
CREATE TABLE freelancer_skills (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    freelancer_id INT UNSIGNED NOT NULL,
    skill_id INT UNSIGNED NOT NULL,
    proficiency_level ENUM('beginner', 'intermediate', 'advanced', 'expert') DEFAULT 'intermediate',
    years_experience DECIMAL(3,1) DEFAULT 1.0,
    verified BOOLEAN DEFAULT FALSE,
    verified_by INT UNSIGNED DEFAULT NULL,
    verified_at DATETIME DEFAULT NULL,
    endorsements_count INT UNSIGNED DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (freelancer_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (skill_id) REFERENCES skills(id) ON DELETE CASCADE,
    FOREIGN KEY (verified_by) REFERENCES users(id) ON DELETE SET NULL,
    UNIQUE KEY unique_freelancer_skill (freelancer_id, skill_id),
    INDEX idx_freelancer_proficiency (freelancer_id, proficiency_level),
    INDEX idx_skill_proficiency (skill_id, proficiency_level),
    INDEX idx_verified (verified)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Skill Endorsements Table
CREATE TABLE skill_endorsements (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    freelancer_skill_id INT UNSIGNED NOT NULL,
    endorsed_by INT UNSIGNED NOT NULL,
    comment VARCHAR(255) DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (freelancer_skill_id) REFERENCES freelancer_skills(id) ON DELETE CASCADE,
    FOREIGN KEY (endorsed_by) REFERENCES users(id) ON DELETE CASCADE,
    UNIQUE KEY unique_endorsement (freelancer_skill_id, endorsed_by),
    INDEX idx_freelancer_skill (freelancer_skill_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Gigs Table
CREATE TABLE gigs (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    freelancer_id INT UNSIGNED NOT NULL,
    title VARCHAR(150) NOT NULL,
    slug VARCHAR(150) DEFAULT NULL,
    description TEXT NOT NULL,
    category_id INT UNSIGNED NOT NULL,
    subcategory_id INT UNSIGNED DEFAULT NULL,
    price DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    pricing_type ENUM('fixed', 'hourly', 'package') DEFAULT 'fixed',
    delivery_time INT UNSIGNED NOT NULL DEFAULT 3 COMMENT 'Delivery time in days',
    revisions_included INT UNSIGNED DEFAULT 0,
    image VARCHAR(255) DEFAULT NULL,
    gallery JSON DEFAULT NULL COMMENT 'Array of image URLs',
    video_url VARCHAR(255) DEFAULT NULL,
    status ENUM('active', 'inactive', 'paused', 'deleted', 'pending_approval') DEFAULT 'active',
    featured BOOLEAN DEFAULT FALSE,
    featured_until DATETIME DEFAULT NULL,
    views_count INT UNSIGNED DEFAULT 0,
    clicks_count INT UNSIGNED DEFAULT 0,
    orders_count INT UNSIGNED DEFAULT 0,
    orders_in_queue INT UNSIGNED DEFAULT 0,
    avg_rating DECIMAL(3,2) DEFAULT 0.00,
    reviews_count INT UNSIGNED DEFAULT 0,
    tags JSON DEFAULT NULL COMMENT 'Array of tags',
    requirements TEXT DEFAULT NULL COMMENT 'What the freelancer needs from client',
    extra_fast_delivery BOOLEAN DEFAULT FALSE,
    extra_fast_delivery_time INT UNSIGNED DEFAULT NULL,
    extra_fast_delivery_fee DECIMAL(10,2) DEFAULT NULL,
    commercial_use BOOLEAN DEFAULT FALSE,
    source_files BOOLEAN DEFAULT FALSE,
    seo_score TINYINT UNSIGNED DEFAULT 0,
    last_modified_at TIMESTAMP NULL DEFAULT NULL,
    approved_at DATETIME DEFAULT NULL,
    approved_by INT UNSIGNED DEFAULT NULL,
    deleted_at DATETIME DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (freelancer_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE CASCADE,
    FOREIGN KEY (subcategory_id) REFERENCES categories(id) ON DELETE SET NULL,
    FOREIGN KEY (approved_by) REFERENCES users(id) ON DELETE SET NULL,
    
    INDEX idx_gig_status_featured (status, featured, avg_rating DESC),
    INDEX idx_gig_category_status (category_id, status),
    INDEX idx_gig_freelancer_status (freelancer_id, status),
    INDEX idx_gig_price_range (status, price),
    INDEX idx_gig_rating_reviews (avg_rating DESC, reviews_count DESC),
    INDEX idx_gig_created (created_at DESC),
    INDEX idx_gig_featured_until (featured, featured_until),
    INDEX idx_deleted (deleted_at),
    INDEX idx_slug_status (slug, status),
    FULLTEXT idx_gig_search (title, description)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 4. ORDERS, JOB REQUESTS & PROJECTS

-- Orders Table
CREATE TABLE orders (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    order_number VARCHAR(50) NOT NULL UNIQUE,
    client_id INT UNSIGNED NOT NULL,
    freelancer_id INT UNSIGNED NOT NULL,
    gig_id INT UNSIGNED NOT NULL,
    package_type ENUM('basic', 'standard', 'premium', 'custom') DEFAULT 'basic',
    status ENUM('pending', 'requirements_pending', 'in_progress', 'delivered', 'revision', 'completed', 'cancelled', 'disputed', 'late') DEFAULT 'pending',
    amount DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    service_fee DECIMAL(10,2) DEFAULT 0.00,
    total_amount DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    delivery_time INT UNSIGNED NOT NULL DEFAULT 3,
    extra_services JSON DEFAULT NULL COMMENT 'Additional services purchased',
    requirements JSON DEFAULT NULL COMMENT 'Client answers to requirements',
    special_instructions TEXT DEFAULT NULL,
    priority ENUM('normal', 'high') DEFAULT 'normal',
    auto_complete_at DATETIME DEFAULT NULL,
    started_at DATETIME DEFAULT NULL,
    delivered_at DATETIME DEFAULT NULL,
    completed_at DATETIME DEFAULT NULL,
    cancelled_at DATETIME DEFAULT NULL,
    cancellation_reason TEXT DEFAULT NULL,
    cancelled_by INT UNSIGNED DEFAULT NULL,
    dispute_reason TEXT DEFAULT NULL,
    dispute_opened_at DATETIME DEFAULT NULL,
    late_delivery BOOLEAN DEFAULT FALSE,
    revision_count INT UNSIGNED DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (client_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (freelancer_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (gig_id) REFERENCES gigs(id) ON DELETE CASCADE,
    FOREIGN KEY (cancelled_by) REFERENCES users(id) ON DELETE SET NULL,
    
    INDEX idx_order_status_created (status, created_at DESC),
    INDEX idx_order_client_status (client_id, status),
    INDEX idx_order_freelancer_status (freelancer_id, status),
    INDEX idx_order_gig (gig_id),
    INDEX idx_order_dates (created_at, completed_at),
    INDEX idx_auto_complete (auto_complete_at),
    INDEX idx_order_number (order_number)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Job Requests Table (Directly used in Client & Freelancer dashboards)
CREATE TABLE job_requests (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    client_id INT UNSIGNED NOT NULL,
    freelancer_id INT UNSIGNED NOT NULL,
    gig_id INT UNSIGNED NOT NULL,

    request_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    completion_date DATETIME DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (client_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (freelancer_id) REFERENCES users(id) ON DELETE CASCADE,

-- Projects Table (Used in Hire Freelancer direct project workflow)

-- Order Status History Table (Audit trail)

    INDEX idx_order_status (order_id, created_at DESC)

-- ============================================================
-- 5. PAYMENTS & WALLETS
-- ============================================================

-- Transactions Table
CREATE TABLE transactions (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    transaction_id VARCHAR(100) NOT NULL UNIQUE,
    order_id INT UNSIGNED DEFAULT NULL,
    user_id INT UNSIGNED NOT NULL,
    transaction_type ENUM('payment', 'refund', 'withdrawal', 'deposit', 'service_fee', 'bonus', 'penalty') NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    currency VARCHAR(10) DEFAULT 'USD',
    status ENUM('pending', 'processing', 'completed', 'failed', 'cancelled', 'refunded') DEFAULT 'pending',
    payment_method ENUM('credit_card', 'paypal', 'stripe', 'bank_transfer', 'wallet', 'other') NOT NULL DEFAULT 'wallet',
    payment_gateway VARCHAR(50) DEFAULT NULL,
    gateway_transaction_id VARCHAR(255) DEFAULT NULL,
    payment_gateway_response TEXT DEFAULT NULL,
    description TEXT DEFAULT NULL,
    metadata JSON DEFAULT NULL,
    ip_address VARCHAR(45) DEFAULT NULL,
    processed_at DATETIME DEFAULT NULL,
    failed_reason TEXT DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE SET NULL,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_transaction_user_type (user_id, transaction_type, status),
    INDEX idx_transaction_order (order_id),
    INDEX idx_transaction_status_created (status, created_at DESC),
    INDEX idx_transaction_gateway (payment_gateway, gateway_transaction_id),
    INDEX idx_trans_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- User Wallets Table
CREATE TABLE wallets (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id INT UNSIGNED NOT NULL UNIQUE,
    balance DECIMAL(15,2) DEFAULT 0.00,
    pending_clearance DECIMAL(15,2) DEFAULT 0.00,
    reserved_amount DECIMAL(15,2) DEFAULT 0.00,
    total_earned DECIMAL(15,2) DEFAULT 0.00,
    total_withdrawn DECIMAL(15,2) DEFAULT 0.00,
    total_spent DECIMAL(15,2) DEFAULT 0.00,
    currency VARCHAR(10) DEFAULT 'USD',
    minimum_withdrawal DECIMAL(10,2) DEFAULT 50.00,
    last_transaction_at DATETIME DEFAULT NULL,
    version INT UNSIGNED DEFAULT 0 COMMENT 'For optimistic locking',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_balance (balance DESC),
    INDEX idx_last_transaction (last_transaction_at DESC)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
